// data/schemas/editorial.ts — imports only 'zod'
export const Fact = z.strictObject({ text: z.string().min(1), source: z.union([z.string().url(), z.enum(['dem', 'exif', 'Tim'])]) });
export const ElevationRef = z.discriminatedUnion('kind', [ z.strictObject({ kind: z.literal('photo') }), z.strictObject({ kind: z.literal('summit') }) ]);
const EditorialAnnotation = z.strictObject({ id: z.string(), title: z.string(), description: z.string(),
  position: z.strictObject({ x: z.number().min(0).max(100), y: z.number().min(0).max(100) }),
  icon: z.enum(['mountain', 'water', 'glacier', 'trail', 'lake', 'peak']),
  details: z.strictObject({ elevation: ElevationRef.optional(), distance: Fact.optional(), facts: z.array(Fact).optional() }).optional() });
export const Editorial = z.strictObject({ slug: z.string().regex(/^[a-z0-9]+(-[a-z0-9]+)*$/), title: z.string(), alt: z.string(),
  location: z.string(), story: z.string(), subjectTags: z.array(z.string().regex(/^[a-z]+(-[a-z]+)*$/)), annotations: z.array(EditorialAnnotation) });
export const EditorialMap = z.record(z.string(), Editorial);
// data/photos/editorial.ts
export const editorial = { 'img-0605': {…}, 'img-0629': {…}, 'img-0662': {…}, 'r004-011': {…} } as const;
