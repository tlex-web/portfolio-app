```typescript
// data/schemas/project.ts — imports only 'zod'
const Scalar = z.union([z.string(), z.number(), z.boolean()]);
MetricsBlock   = z.strictObject({ kind: z.literal('metrics'),    key: z.string(), title: z.string(), metrics: z.array(z.strictObject({ key: z.string(), label: z.string(), value: Scalar })).min(1) });
RiskScaleBlock = z.strictObject({ kind: z.literal('risk-scale'), key: z.string(), title: z.string(), rules: z.number().int(),
                   bands: z.array(z.strictObject({ level: RiskLevel, min: z.number(), max: z.number() })).length(4),
                   flags: z.array(z.strictObject({ key: z.string(), label: z.string(), value: z.boolean() })) });
PricingBlock   = z.strictObject({ kind: z.literal('pricing'), key: z.string(), title: z.string(), tiers: z.array(z.strictObject({ key: z.string(), price: z.string(), commands: z.string(), models: z.string(), features: z.array(z.string()) })).min(1) });
ListBlock      = z.strictObject({ kind: z.literal('list'),  key: z.string(), title: z.string(), items: z.array(z.string()).min(1) });
StackBlock     = z.strictObject({ kind: z.literal('stack'), key: z.string(), title: z.string(), groups: z.array(z.strictObject({ key: z.string(), items: z.array(z.string()).min(1) })).min(1) });
export const HighlightBlock = z.discriminatedUnion('kind', [MetricsBlock, RiskScaleBlock, PricingBlock, ListBlock, StackBlock]);
export const DemoCommandSchema;  // Task 1: mirrors today's interface; Task 3: riskLevel and riskScore (0..1) required
const ClixCli     = ProjectBase.extend({ slug: z.literal('clix-cli'),     highlights: z.tuple([MetricsBlock, MetricsBlock, RiskScaleBlock, MetricsBlock]), demoCommands: z.array(DemoCommandSchema).min(1) });
const ClixWebsite = ProjectBase.extend({ slug: z.literal('clix-website'), highlights: z.tuple([PricingBlock, ListBlock, StackBlock]) });
export const ProjectSchema = z.discriminatedUnion('slug', [ClixCli, ClixWebsite]);

// lib/legacy-highlights.ts
export function toLegacyHighlights(blocks: readonly HighlightBlockT[]): Record<string, HighlightValue>;
//  metrics    -> { [key]: { [m.key]: m.value, ... } }
//  risk-scale -> { [key]: { rules, riskLevels: bands.map(b => `${Cap(level)} (${min.toFixed(1)}-${max.toFixed(1)})`), ...flags as key: value } } in that key order
//  pricing    -> { [key]: { [tier.key]: { price, commands, models, features } } }
//  list       -> { [key]: items }
//  stack      -> { [key]: { [group.key]: items } }

// lib/trail-grade.ts — zero imports
export const RISK_BANDS: readonly { level: 'low'|'medium'|'high'|'critical'; min: number; max: number; grade: string; marker: string }[];
export function bandFor(score: number): (typeof RISK_BANDS)[number];
export function isScoreInLevel(score: number, level: RiskLevel): boolean;
```
