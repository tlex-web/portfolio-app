export type Wgs84 = { lat: number; lon: number };

export function stripBbox(extent: Bbox, summitN: number, ratio?: number, snapM?: number): Bbox; // defaults 5, 100

export type Pt = [number, number];
export function onEdge(p: Pt, q: Pt, W: number, H: number): boolean;

export function roundHalf(v: number): number;
export function runToPath(points: { x: number; y: number }[], closed: boolean): string; // "M x yLx y...[Z]"

// lib/terrain/hillshade.ts
export function hillshade(h: Float32Array, W: number, H: number, cellM: number, azDeg?: number, altDeg?: number, z?: number): Float32Array;
export function blendShades(a: Float32Array, b: Float32Array, weightA: number): Float32Array;

- **DATA-01 (unclassified edge row):** "Zermatt–Matterhorn area" is the D-14 extent chosen at the 08-04 checkpoint. These cores take the extent as a parameter, so a different extent changes data only, never this code.

Visual review in 08-13 catches an engine difference.
