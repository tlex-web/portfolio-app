```typescript
export const Capture = z.discriminatedUnion('precision', [
  z.strictObject({ precision: z.literal('second'), local: NaiveLocal, timeZone: z.literal('Europe/Zurich'), offset: z.enum(['+01:00', '+02:00']), source: z.literal('exif') }),
  z.strictObject({ precision: z.literal('year'), year: z.number().int().min(1950).max(2100), source: z.literal('Tim') }),
]);
export const Position = z.strictObject({ lat: Coord4, lon: Coord4, source: z.enum(['exif', 'hand-placed']), approximate: z.boolean(),
  reason: z.string().min(1).optional(), radiusM: z.number().positive().optional() })
  .refine(p => !p.approximate || (p.reason && p.radiusM), 'approximate positions need reason + radiusM (DATA-04)')
  .refine(p => p.source === 'exif' || p.approximate, 'hand-placed positions are approximate (DATA-04)');
export const Elevation = z.discriminatedUnion('source', [ z.strictObject({ source: z.literal('dem'), m: z.number().int() }) ]);
export const Exposure = z.strictObject({ camera: z.string(), lens: z.string(), focalLengthMm: z.number(), fNumber: z.number(), exposureTimeS: z.number(), iso: z.number().int() });
export const LocationOverride = z.strictObject({ name: z.string().min(1), reason: z.string().min(1), source: z.literal('Tim') });
export const Corrections = z.strictObject({ schemaVersion: z.literal(1), photos: z.record(z.string(), z.strictObject({
  medium: z.enum(['digital', 'film-scan']).optional(),
  captureOverride: z.strictObject({ precision: z.literal('year'), year: z.number().int(), source: z.literal('Tim'), reason: z.string().min(1) }).optional(),
  position: Position.optional(), locationOverride: LocationOverride.optional() })) });
export const PhotoFacts = z.strictObject({ id, medium: z.enum(['digital', 'film-scan']), capture: Capture, position: Position.nullable(),
  uv: z.strictObject({ u: z.number(), v: z.number() }).nullable(), elevation: Elevation.nullable(), exif: Exposure.nullable(),
  tags: z.strictObject({ year: z.string().regex(/^\d{4}$/), season: z.enum(['winter', 'spring', 'summer', 'autumn']).nullable() }),
  locationOverride: LocationOverride.nullable() });
```
